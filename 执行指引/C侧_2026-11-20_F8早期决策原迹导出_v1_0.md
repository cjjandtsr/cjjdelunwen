## 步骤（WSL bash）
```bash
E=/home/cjj/AUV/program/export/C侧_20261120_f8_early
R3=/home/cjj/AUV/program/results/routea_f8_diag_run03
R1=/home/cjj/AUV/program/results/routea_f8_run01
mkdir -p "$E"

# 1. 目录结构（用于核对路径；建库原迹若不在下面第 3 步的匹配范围内，以此为准回报）
find "$R3" -maxdepth 3 -type d > "$E/run03_dirs.txt"
find "$R1" -maxdepth 4 -type d > "$E/run01_dirs.txt"

# 2. D1 五臂原迹：每份打印文件名，再打印前 7 行（表头 + 前 6 次决策）
for A in P6 P20 P21 P22 P23; do
  find "$R3/d1/$A/traces" -name '*.jsonl' -print -exec head -n 7 {} \; > "$E/d1_${A}_head7.txt"
done

# 3. 建库飞行原迹：同样前 7 行（路径中含 collect 的 jsonl）
find "$R1" -path '*collect*' -name '*.jsonl' -print -exec head -n 7 {} \; > "$E/collection_head7.txt"
find "$R1" -path '*collect*' -name '*.csv' -exec cp --parents {} "$E/" \;

# 4. 计数与校验
ls -la "$E" > "$E/ls.txt"
find "$E" -type f ! -name SHA256SUMS -exec sha256sum {} \; > "$E/SHA256SUMS"

# 5. 打包回传
tar -C /home/cjj/AUV/program/export -czf /home/cjj/AUV/program/export/C侧_20261120_f8_early.tar.gz C侧_20261120_f8_early
sha256sum /home/cjj/AUV/program/export/C侧_20261120_f8_early.tar.gz
```
