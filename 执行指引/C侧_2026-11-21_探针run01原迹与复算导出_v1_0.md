## 命令（WSL）
```bash
R=/home/cjj/AUV/program/results/routea_probe_e3g_run01
V=/mnt/c/Users/cjj/Desktop/IDP3/program/e3g_review_20261121_complete
O=/home/cjj/AUV/program/jindu/C侧_20261121_probe_export
mkdir -p "$O"
# 1. 探针运行的目录结构与 traces 目录文件数
find "$R" -maxdepth 5 -type d | sort > "$O/tree.txt"
find "$R" -type d -name traces | sort > "$O/trace_dirs_all.txt"
for d in $(cat "$O/trace_dirs_all.txt"); do echo "$d $(ls "$d" | wc -l)"; done > "$O/trace_counts.txt"
# 2. 只取正式评价（#E3、G0、G1 的 dev420 与 val）的 traces；不含 smoke、网格、被杀试次
grep -i "formal" "$O/trace_dirs_all.txt" | grep -vi "smoke\|attempt01\|grid" > "$O/trace_dirs_formal.txt"
cd / && tar -czf "$O/probe_run01_formal_traces.tar.gz" $(sed 's#^/##' "$O/trace_dirs_formal.txt")
# 3. Codex 复算导出整目录
cd "$(dirname "$V")" && tar -czf "$O/e3g_review_20261121_complete.tar.gz" "$(basename "$V")"
# 4. 定位 BC、P6 在论文 186 任务（银行 d34c10d6…）上的评价运行：只列目录与首行
find /home/cjj/AUV/program/results -maxdepth 1 -type d -iname "*paper*" | sort > "$O/paper_runs.txt"
for r in $(cat "$O/paper_runs.txt"); do for d in $(find "$r" -maxdepth 4 -type d -name traces); do f=$(ls -d "$d"/* | head -1); echo "== $d $(ls "$d" | wc -l)"; head -c 1500 "$f"; echo; done; done > "$O/paper_runs_heads.txt"
# 5. 哈希
sha256sum "$O"/* > "$O/SHA256SUMS"; echo "rc=$?" >> "$O/commands_rc.log"; ls -la "$O"
```

第 2 步的筛选词以 run01 的实际目录名为准。若正式评价目录名里不含 `formal`，先回传 `tree.txt` 与 `trace_counts.txt`，由 Claude 指定目录。
